import logging

from aiogram import F, Router
from aiogram.filters import Command, CommandStart
from aiogram.types import CallbackQuery, InlineKeyboardMarkup, Message
from aiogram.utils.keyboard import InlineKeyboardBuilder

from app.ocr_utils import answer_from_text, extract_text_from_image, extract_text_from_pdf
from app.topics import TOPICS

router = Router()
logger = logging.getLogger(__name__)


def build_topics_keyboard() -> InlineKeyboardMarkup:
    builder = InlineKeyboardBuilder()
    for topic_id, topic in TOPICS.items():
        builder.button(text=f"{topic_id}. {topic['title'][:40]}", callback_data=f"topic:{topic_id}")
    builder.adjust(1)
    return builder.as_markup()


def format_topic(topic_id: int) -> str:
    topic = TOPICS[topic_id]
    return (
        f"<b>{topic['title']}</b>\n\n"
        f"{topic['summary']}\n\n"
        f"<b>{topic['answer']}</b>"
    )


@router.message(CommandStart())
async def start(message: Message) -> None:
    await message.answer(
        "Assalomu alaykum! Bu bot Diskret tuzilmalar fanini o'rganishga yordam beradi.\n\n"
        "Mavzuni tanlang yoki /topics buyrug'ini yuboring. "
        "Savol, rasm yoki PDF yuborsa, men matnni o'qib tegishli mavzuni aniqlayman.",
        reply_markup=build_topics_keyboard(),
    )


@router.message(Command("help"))
async def help_command(message: Message) -> None:
    await message.answer(
        "Buyruqlar:\n"
        "/start — botni boshlash\n"
        "/topics — mavzular ro'yxati\n"
        "/help — yordam\n\n"
        "Mavzuni raqami, menyudagi tugma yoki matnli savol orqali tanlang."
    )


@router.message(Command("topics"))
async def topics_command(message: Message) -> None:
    await message.answer("Diskret tuzilmalar mavzulari:", reply_markup=build_topics_keyboard())


@router.callback_query(F.data.startswith("topic:"))
async def topic_callback(callback: CallbackQuery) -> None:
    try:
        topic_id = int((callback.data or "").split(":", maxsplit=1)[1])
    except (IndexError, ValueError):
        await callback.answer("Mavzu raqami noto'g'ri.", show_alert=True)
        return

    if topic_id not in TOPICS:
        await callback.answer("Bunday mavzu mavjud emas.", show_alert=True)
        return

    if callback.message:
        await callback.message.answer(format_topic(topic_id), reply_markup=build_topics_keyboard())
    await callback.answer()


@router.message(F.photo)
async def photo_message(message: Message) -> None:
    photo = message.photo[-1]
    file = await message.bot.get_file(photo.file_id)
    image_bytes = await message.bot.download_file(file.file_path)

    await message.answer("Rasm o'qilmoqda. Iltimos, biroz kuting...")
    extracted = extract_text_from_image(image_bytes)

    if not extracted:
        await message.answer(
            "Rasmdan matn o'qib bo'lmadi. Mavzuni tanlang yoki savolni matn shaklida yuboring."
        )
        return

    await message.answer(answer_from_text(extracted))


@router.message(F.document)
async def document_message(message: Message) -> None:
    doc = message.document
    if not doc:
        return

    file = await message.bot.get_file(doc.file_id)
    payload = await message.bot.download_file(file.file_path)

    await message.answer("Fayl o'qilmoqda. Iltimos, biroz kuting...")
    ext = (doc.file_name or "").lower()

    if ext.endswith(".pdf"):
        extracted = extract_text_from_pdf(payload)
    else:
        extracted = ""

    if not extracted:
        await message.answer(
            "Fayldan matn o'qib bo'lmadi. Mavzuni tanlang yoki matnli savol yuboring."
        )
        return

    await message.answer(answer_from_text(extracted))


@router.message(F.text)
async def text_message(message: Message) -> None:
    text = (message.text or "").strip()
    if text.isdecimal():
        topic_id = int(text)
        if topic_id in TOPICS:
            await message.answer(format_topic(topic_id), reply_markup=build_topics_keyboard())
            return

    if text:
        await message.answer(answer_from_text(text))
        return

    await message.answer(
        "Mavzuni menyudan tanlang yoki raqamini yuboring (1-14).\n"
        "Mavzular uchun /topics buyrug'ini yuboring."
    )
